  11: def train_epoch(model, dataset, optimizer, loss_fn, device):
  12:     model.train()
  13:     current_loss = 0.0
  14:     for batch in dataset:
  15:         x,y = batch
  16:         x = x.to(device)
  17:         y = y.to(device)
  18: 
  19:         if model.type == 'simple_mlp' or model.type == 'dropout':
  20:             pred = model(x)
  21:             loss = loss_fn(pred, y)
  22: 
  23:         elif model.type == 'zigzag':
  24:             pred0 = model(x)
  25:             pred1 = model(x,y )
  26:             if new_zz: loss = loss_fn(pred0,y) + loss_fn(pred1, torch.zeros_like(y)) # new zigzag
  27:             else: loss = loss_fn(pred0,y) + loss_fn(pred1, y) # original zigzag
  28: 
  29:         optimizer.zero_grad()
  30:         loss.backward()
  31:         optimizer.step()
  32: 
  33:         current_loss += loss.item()
  34: 
  35:     return current_loss / len(dataset), loss
  36: 
  37: def test_epoch(model, dataset, metrics, device):
  38:     model.eval()
  39:     with torch.no_grad():
  40:         res_list = [0. for metric in metrics]
  41:         N = len(dataset)
  42:         for batch in dataset:
  43:             x,y = batch
  44:             x = x.to(device)
  45:             y = y.to(device)
  46:             pred = model(x)
  47:             for i, metric in enumerate(metrics):
  48:                 res_list[i] += metric(pred, y)/N
